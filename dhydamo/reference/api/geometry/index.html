    def as_linestring_list(
    linestring: LineString | MultiLineString | list[LineString | MultiLineString],
) -> list[LineString]:
    """Return the LineStrings in a LineString, a MultiLineString or a list of those.

    Useful to iterate over geometries that can be single or multi.

    Parameters
    ----------
    linestring : LineString, MultiLineString or list of those
        Geometry, or geometries, to split.

    Returns
    -------
    list[LineString]
        The single LineStrings, in order.

    Raises
    ------
    TypeError
        If a geometry is not a LineString or MultiLineString.
    """
    if isinstance(linestring, LineString):
        return [linestring]
    if isinstance(linestring, MultiLineString):
        return list(linestring.geoms)
    if not isinstance(linestring, list):
        raise TypeError(
            f'Expected LineString or MultiLineString. Got "{type(linestring)}"'
        )
    linestrings: list[LineString] = []
    for item in linestring:
        linestrings.extend(as_linestring_list(item))
    return linestrings

    def interp_linestring(
    linestring: LineString | MultiLineString, dist: float
) -> LineString | MultiLineString:
    """Resample a line at regular distances along it.

    Parameters
    ----------
    linestring : LineString or MultiLineString
        Line to resample. The parts of a MultiLineString are resampled one by one.
    dist : float
        Desired distance between the vertices. The line gets ``round(length / dist)``
        vertices, at least 4, evenly spread from start to end.

    Returns
    -------
    LineString or MultiLineString
        The resampled line, of the same type as the input.
    """
    # In case of multilinestring
    if isinstance(linestring, MultiLineString):
        return MultiLineString([interp_linestring(ls, dist) for ls in linestring.geoms])

    n_points = max(4, round(linestring.length / dist))
    distances = np.linspace(0, linestring.length, n_points)
    points = shapely.line_interpolate_point(linestring, distances)
    return LineString(points)

    def interp_polygon(
    polygon: Polygon | MultiPolygon, dist: float
) -> Polygon | MultiPolygon:
    """Simplify a polygon, then resample its exterior and holes at regular distances.

    Parameters
    ----------
    polygon : Polygon or MultiPolygon
        Polygon to resample. The parts of a MultiPolygon are resampled one by one.
    dist : float
        Tolerance for the simplification, and desired distance between the vertices,
        see `interp_linestring`.

    Returns
    -------
    Polygon or MultiPolygon
        The resampled polygon, of the same type as the input.

    Raises
    ------
    TypeError
        If simplifying the polygon does not give a Polygon.
    """
    # In case of multipolygon
    if isinstance(polygon, MultiPolygon):
        return MultiPolygon([interp_polygon(p, dist) for p in polygon.geoms])

    simplified = polygon.simplify(dist)
    if not isinstance(simplified, Polygon):
        raise TypeError(f"Simplifying a Polygon gave a {simplified.geom_type}")
    polygon = Polygon(
        shell=interp_linestring(simplified.exterior, dist),
        holes=[interp_linestring(intr, dist) for intr in simplified.interiors],
    )
    return polygon

    def links1d2d_add_links_1d_to_2d(
    network: Network,
    branchids: list[str] | None = None,
    within: Polygon | MultiPolygon | None = None,
    max_length: float = np.inf,
) -> None:
    """Add 1D2D links to the network, generated from the 1D nodes to the 2D faces.

    Parameters
    ----------
    network : Network
        Network in which the links are made.
    branchids : list[str], optional
        Branches to link. Defaults to None: all branches.
    within : Polygon or MultiPolygon, optional
        Area within which links are made. Defaults to None: the extent of both meshes.
    max_length : float, optional
        Maximum length of a link. Defaults to ``np.inf``.
    """
    # Fetch the 2d mesh once and reuse it below - each network._mesh2d.mesh2d_*
    # property otherwise re-fetches and re-validates the whole mesh from meshkernel.
    mesh2d_output = network._mesh2d.get_mesh2d()

    if within is None:
        # If not provided, create a box from the maximum bounds
        xmin = min(network._mesh1d.mesh1d_node_x.min(), mesh2d_output.node_x.min())
        xmax = max(network._mesh1d.mesh1d_node_x.max(), mesh2d_output.node_x.max())
        ymin = min(network._mesh1d.mesh1d_node_y.min(), mesh2d_output.node_y.min())
        ymax = max(network._mesh1d.mesh1d_node_y.max(), mesh2d_output.node_y.max())

        within = box(xmin, ymin, xmax, ymax)

    # If a 'within' polygon was provided, convert it to a geometrylist
    geometrylist = _geomlist_from_polygon(within)

    # Get the nodes for the specific branch ids
    node_mask = network._mesh1d.get_node_mask(branchids)

    # Snapshot the already present links. They are restored after the call below
    # regardless of whether that call preserves them itself (see _filter_links_on_idx).
    present_links = network._link1d2d.link1d2d

    # Generate links
    network._link1d2d._link_from_1d_to_2d(node_mask, polygon=geometrylist)

    # Filter the newly generated links that are longer than the max distance.
    # Links that were already present are never subject to this filter.
    current_links = network._link1d2d.link1d2d
    # TODO: fix in hydrolib-core, link1d2d is annotated np.ndarray[np.int32] (int32 read as shape):
    # https://github.com/Deltares/HYDROLIB-core/issues/1314
    is_new = _new_link_mask(current_links, present_links)  # type: ignore[arg-type]
    id1d = current_links[is_new, 0]
    id2d = current_links[is_new, 1]
    nodes1d = np.stack(
        [network._mesh1d.mesh1d_node_x[id1d], network._mesh1d.mesh1d_node_y[id1d]],
        axis=1,
    )
    faces2d = np.stack(
        [mesh2d_output.face_x[id2d], mesh2d_output.face_y[id2d]],
        axis=1,
    )
    lengths = np.hypot(nodes1d[:, 0] - faces2d[:, 0], nodes1d[:, 1] - faces2d[:, 1])
    keep_mask = ~is_new
    keep_mask[is_new] = lengths < max_length
    keep = np.nonzero(keep_mask)[0]
    # TODO: fix in hydrolib-core, link1d2d is annotated np.ndarray[np.int32] (int32 read as shape):
    # https://github.com/Deltares/HYDROLIB-core/issues/1314
    _filter_links_on_idx(network, keep, present_links)  # type: ignore[arg-type]

    def links1d2d_add_links_2d_to_1d_embedded(
    network: Network,
    branchids: list[str] | None = None,
    within: Polygon | MultiPolygon | None = None,
) -> None:
    """Add 1D2D links where the 2D mesh overlaps the 1D mesh: the 'embedded' links.

    Each 2D face that crosses a branch is linked to the 1D mesh. To find these faces
    efficiently:

    1. get the maximum length of a face edge;
    2. buffer the branches with this length;
    3. find the face centers within this buffer;
    4. check for each of these faces if it crosses the branches.

    Parameters
    ----------
    network : Network
        Network in which the links are made, with a 1D and a 2D mesh.
    branchids : list[str], optional
        Branches to link. Defaults to None: all branches.
    within : Polygon or MultiPolygon, optional
        Area within which links are made. Defaults to None: everywhere.

    Raises
    ------
    TypeError
        If clipping the buffered branches with ``within`` does not give a polygon.
    """
    # Fetch the 2d mesh once and reuse it below - each network._mesh2d.mesh2d_*
    # property otherwise re-fetches and re-validates the whole mesh from meshkernel.
    mesh2d_output = network._mesh2d.get_mesh2d()

    # Get the max edge distance
    nodes2d = np.stack([mesh2d_output.node_x, mesh2d_output.node_y], axis=1)
    edge_node_crds = nodes2d[network._mesh2d.mesh2d_edge_nodes]

    diff = edge_node_crds[:, 0, :] - edge_node_crds[:, 1, :]
    maxdiff = np.hypot(diff[:, 0], diff[:, 1]).max()

    # Create multilinestring from branches
    nodes1d = np.stack(
        [network._mesh1d.mesh1d_node_x, network._mesh1d.mesh1d_node_y], axis=1
    )

    # Buffer the branches with the max cell distances
    mls = MultiLineString(nodes1d[network._mesh1d.mesh1d_edge_nodes].tolist())
    area: Polygon | MultiPolygon = mls.buffer(maxdiff)

    # If a within polygon is provided, clip the buffered area with this polygon.
    if within is not None:
        clipped = area.intersection(within)
        if not isinstance(clipped, (Polygon, MultiPolygon)):
            raise TypeError(
                f"Expected the clipped area to be a Polygon or MultiPolygon, not a {clipped.geom_type}."
            )
        area = clipped

    # Create an array with 2d facecenters and check which intersect the (clipped) area
    faces2d = np.stack([mesh2d_output.face_x, mesh2d_output.face_y], axis=1)
    mpgl = mk.GeometryList(*faces2d.T.copy())
    idx = np.zeros(len(faces2d), dtype=bool)

    for subarea in common.as_polygon_list(area):
        subarea = GeometryList.from_polygon(subarea)
        sub_idx = network.meshkernel.polygon_get_included_points(subarea, mpgl)
        # values of a meshkernel GeometryList, not pandas
        idx |= sub_idx.values == 1  # noqa: PD011

    # Check for each of the remaining faces, if it actually crosses the branches
    where = np.nonzero(idx)[0]

    # Return early if there are no candidates
    if not where.size:
        return

    # Check which face crosses a branch
    candidate_face_nodes = network._mesh2d.mesh2d_face_nodes[idx]
    face_polygons = []
    for face_idxs in candidate_face_nodes:
        # nodes2d always has 4 columns if the grid is a mix of triangular and
        # rectangular, but a triangular cell only has 3 nodes. A triangular
        # cell will then have a column of invalid indices. This needs to be
        # checked per cell.
        # Use a polygon instead of linestring to enforce a closed ring and to
        # test both boundary and/or face intersection.
        valid_face_idxs = face_idxs[face_idxs >= 0]
        face_polygons.append(Polygon(nodes2d[valid_face_idxs]))
    idx[where] = shapely.intersects(face_polygons, mls)

    # Use the remaining points to create the links
    multipoint = mk.GeometryList(
        x_coordinates=faces2d[idx, 0], y_coordinates=faces2d[idx, 1]
    )

    # Get the nodes for the specific branch ids
    node_mask = network._mesh1d.get_node_mask(branchids)

    # Snapshot the already present links. They are restored after the call below
    # regardless of whether that call preserves them itself (see _filter_links_on_idx).

    def links1d2d_add_links_2d_to_1d_lateral(
    network: Network,
    dist_factor: float | None = 2.0,
    branchids: list[str] | None = None,
    within: Polygon | MultiPolygon | None = None,
    max_length: float = np.inf,
) -> None:
    """Add lateral 1D2D links, from the 2D mesh boundary to the 1D mesh beside it.

    Whether a link is kept depends on the distance from the face center to where the
    link crosses the boundary of the 2D mesh. By default, links that cross further than
    2 times the mean distance from the face center to its nodes are removed. For a
    square cell with a direct link out of the cell (without passing other cells) this
    ratio is at most sqrt(2) = 1.414; the default of 2 gives some margin. A link that
    crosses the 2D mesh boundary more than once is always removed.

    Parameters
    ----------
    network : Network
        Network in which the links are made, with a 1D and a 2D mesh.
    dist_factor : float, optional
        Factor that determines which links are kept, see above. Defaults to 2.0. With
        None, all links are kept.
    branchids : list[str], optional
        Branches to link. Defaults to None: all branches.
    within : Polygon or MultiPolygon, optional
        Area within which 2D faces are linked. Defaults to None: everywhere.
    max_length : float, optional
        Maximum length of a link. Defaults to ``np.inf``.

    Raises
    ------
    TypeError
        If the mesh boundaries, or their intersection with ``within``, are not polygons.
    """
    geometrylist = network.meshkernel.mesh2d_get_mesh_boundaries_as_polygons()
    mpboundaries = GeometryList(**geometrylist.__dict__).to_geometry()
    if not isinstance(mpboundaries, (Polygon, MultiPolygon)):
        raise TypeError(
            f"Expected the mesh boundaries to be polygons, not a {mpboundaries.geom_type}."
        )
    if within is not None:
        # If a 'within' polygon was provided, get the intersection with the meshboundaries
        # and convert it to a geometrylist
        # Note that the provided meshboundaries is a (list of) polygon(s). Holes are provided
        # as polygons as well, which dont make it a valid MultiPolygon
        clipped: list[Polygon | MultiPolygon] = []
        for boundary in common.as_polygon_list(mpboundaries):
            part = boundary.intersection(within)
            if not isinstance(part, (Polygon, MultiPolygon)):
                raise TypeError(
                    f"Expected the clipped mesh boundary to be a Polygon or MultiPolygon, not a {part.geom_type}."
                )
            clipped.append(part)
        geometrylist = GeometryList.from_geometry(
            MultiPolygon(common.as_polygon_list(clipped))
        )

    # Get the nodes for the specific branch ids
    node_mask = network._mesh1d.get_node_mask(branchids)

    # Snapshot the already present links. They are restored after the call below
    # regardless of whether that call preserves them itself (see _filter_links_on_idx).
    present_links = network._link1d2d.link1d2d

    # Generate links
    network._link1d2d._link_from_2d_to_1d_lateral(
        node_mask, polygon=geometrylist, search_radius=max_length
    )

    # If the provided distance factor was None, no further selection is needed, all links are kept.
    if dist_factor is None:
        return

    # Create multilinestring
    multilinestring = MultiLineString(
        [poly.exterior for poly in common.as_polygon_list(mpboundaries)]
    )

    # Find the links that intersect the boundary close to the origin
    id1d = network._link1d2d.link1d2d[:, 0]
    id2d = network._link1d2d.link1d2d[:, 1]

    nodes1d = np.stack(
        [network._mesh1d.mesh1d_node_x[id1d], network._mesh1d.mesh1d_node_y[id1d]],
        axis=1,
    )
    # Fetch the 2d mesh once and reuse it below - each network._mesh2d.mesh2d_*
    # property otherwise re-fetches and re-validates the whole mesh from meshkernel.
    mesh2d_output = network._mesh2d.get_mesh2d()
    faces2d = np.stack(
        [mesh2d_output.face_x[id2d], mesh2d_output.face_y[id2d]],
        axis=1,
    )
    nodes2d = np.stack([mesh2d_output.node_x, mesh2d_output.node_y], axis=1)
    nodes2d_idx = network._mesh2d.mesh2d_face_nodes[id2d]
    keep = []
    for i, (node1d, face2d, node2d) in enumerate(zip(nodes1d, faces2d, nodes2d_idx)):
        # nodes2d always has 4 columns if the grid is a mix of triangular and
        # rectangular, but a triangular cell only has 3 nodes. A triangular
        # cell will then have a column of invalid indices. This needs to be
        # checked per cell.
        face_node_crds = nodes2d[node2d[node2d >= 0]]

        # Calculate distance between face edge and face center
        x1 = face_node_crds[:, 0]
        y1 = face_node_crds[:, 1]
        x0, y0 = face2d[0], face2d[1]
        distance = np.hypot(x1 - x0, y1 - y0).mean()

        isect = multilinestring.intersection(LineString([face2d, node1d]))

        # If the intersection is for some reason not a Point of Multipoint, skip it.
        if not isinstance(isect, (Point, MultiPoint)):
            continue

        # Skip the link if it has more than one intersection with the boundary
        isect_list = common.as_point_list(isect)
        if len(isect_list) != 1:
            continue

        # If the distance to the mesh 2d exterior intersection is smaller than
        # the compared distance, keep it.
        dist = np.hypot(*(face2d - isect_list[0].coords[:][0]))
        if dist < dist_factor * distance:
            keep.append(i)

    # Select the remaining links
    # TODO: fix in hydrolib-core, link1d2d is annotated np.ndarray[np.int32] (int32 read as shape):
    # https://github.com/Deltares/HYDROLIB-core/issues/1314
    _filter_links_on_idx(network, keep, present_links=present_links)  # type: ignore[arg-type]

                    "%d structures are not linked to a branch (too many to show).",
                    n_missing,

                    n_missing,

        for branchid, group in ids_offsets.groupby("branchid")[["chainage", "id"]]:

    def mesh2d_altitude_from_raster(
    network: Network,
    rasterpath: Path | str,
    where: RasterStatPosition | str = "face",
    stat: str = "mean",
    fill_option: FillOption | str = "fill_value",
    fill_value: float | None = None,
    *,
    min_coverage_frac: float = 0.0,
    coverage_weight: zonal.CoverageWeight = "fraction",
) -> None:
    """Set the level of the 2D faces or nodes from a raster.

    For faces, the statistic is taken over the face polygons; for nodes, over the
    Voronoi cells around the nodes, clipped to the mesh.

    Note that the raster is not clipped. Any values outside the bounds are
    also taken into account.

    Parameters
    ----------
    network : Network
        Network for which the level is determined, with a 2D mesh.
    rasterpath : Path or str
        Raster file from which the levels are read.
    where : RasterStatPosition or str, optional
        Set the level of the faces (``"face"``) or of the nodes (``"node"``). Defaults
        to ``"face"``.
    stat : str, optional
        exactextract statistic of the raster cells per face or node, such as ``"mean"``,
        ``"median"`` or ``"quantile(q=0.9)"``. Defaults to ``"mean"``.
    fill_option : FillOption or str, optional
        How to fill faces or nodes without a raster value: ``"fill_value"``,
        ``"nearest"`` or ``"interpolate"``. Defaults to ``"fill_value"``.
    fill_value : float, optional
        Value for ``"fill_value"``, and for values that cannot be interpolated with
        ``"interpolate"``. Defaults to None.
    min_coverage_frac : float, optional
        Smallest fraction of a raster cell a face or node polygon must cover for the
        cell to count. Defaults to 0: every touched cell.
    coverage_weight : str, optional
        ``"fraction"`` counts raster cells by the fraction of them the polygon covers,
        ``"none"`` counts them fully. Defaults to ``"fraction"``.

    Raises
    ------
    ValueError
        If all values are missing with ``"nearest"`` or ``"interpolate"``, or if
        ``fill_value`` is needed but not given.
    """
    if isinstance(fill_option, str):
        fill_option = FillOption(fill_option)

    if isinstance(where, str):
        where = RasterStatPosition(where)

    # Select points on faces or nodes
    logger.info("Creating GeoDataFrame of cell faces.")

    # Fetch the 2d mesh once and reuse it below - each network._mesh2d.mesh2d_*
    # property otherwise re-fetches and re-validates the whole mesh from meshkernel.
    mesh2d_output = network._mesh2d.get_mesh2d()

    # Get coordinates where z-values are derived or centered
    xy = np.stack(
        [
            getattr(mesh2d_output, f"{where.value}_x"),
            getattr(mesh2d_output, f"{where.value}_y"),
        ],
        axis=1,
    )

    # Create cells as polygons
    xy_facenodes = np.stack([mesh2d_output.node_x, mesh2d_output.node_y], axis=1)
    # Use raw MeshKernel connectivity instead of mesh2d_face_nodes, whose
    # fixed-width rows contain padding for triangles and other shorter faces.
    # nodes_per_face marks each face boundary in the flattened coordinates.
    face_node_counts = mesh2d_output.nodes_per_face
    face_coords = xy_facenodes[mesh2d_output.face_nodes]
    # Shapely's ``indices`` groups coordinates into rings. Repeat each face ID
    # by its node count so every coordinate is assigned to the correct face.
    face_ids = np.arange(len(face_node_counts))
    face_idx = np.repeat(face_ids, face_node_counts)
    facedata = gpd.GeoDataFrame(
        geometry=shapely.polygons(shapely.linearrings(face_coords, indices=face_idx))
    )

    if where == RasterStatPosition.NODE:
        logger.info(
            "Generating voronoi polygons around cell centers for determining raster statistics."
        )
        facedata = spatial.get_voronoi_around_nodes(xy, facedata)

    df = zonal.zonal_stats(
        facedata,
        rasterpath,
        stat,
        strategy="raster-sequential",
        min_coverage_frac=min_coverage_frac,
        coverage_weight=coverage_weight,
    )
    # By position: exactextract names e.g. "quantile(q=0.9)" "quantile_90".
    zvalues = df.iloc[:, 0].to_numpy()

    isnan = np.isnan(zvalues)
    if isnan.any():
        # If interpolation or fill_option, but all are NaN, raise error.
        if isnan.all() and fill_option in [FillOption.NEAREST, FillOption.INTERPOLATE]:
            raise ValueError(
                "Only NaN values found, interpolation or nearest not possible."
            )

        # Fill fill_option values
        if fill_option == FillOption.FILL_VALUE:
            if fill_value is None:
                raise ValueError("Provide a fill_value (keyword argument).")
            # With default value
            zvalues[isnan] = fill_value

        elif fill_option == FillOption.NEAREST:
            # By looking for the nearest value in the grid
            # Create a KDTree of the known points
            tree = KDTree(data=xy[~isnan])
            idx = tree.query(x=xy[isnan])[1]
            zvalues[isnan] = zvalues[~isnan][idx]

        elif fill_option == FillOption.INTERPOLATE:
            if fill_value is None:
                raise ValueError(
                    "Provide a fill_value (keyword argument) to fill values that cannot be interpolated."
                )
            # By interpolating
            isnan = np.isnan(zvalues)
            interp = LinearNDInterpolator(
                xy[~isnan], zvalues[~isnan], fill_value=fill_value
            )
            zvalues[isnan] = interp(xy[isnan])

    # Set values to mesh geometry
    setattr(network._mesh2d, f"mesh2d_{where.value}_z", zvalues)

    face_z = network._mesh2d.mesh2d_face_z

    # value when there is no bed level, and reads them back as that value. Such a bed
    # level is no bed level: keep the network's own instead.

    # faces from the edges and closes small holes into extra faces; even when faces are
    # given, MeshKernel (8.4) puts them in its own order. A face bed level, from the file
    # or of the network, does not follow:
    # https://github.com/Deltares/HYDROLIB-core/issues/1318 (order)
    # https://github.com/Deltares/HYDROLIB-core/issues/577 (extra faces)
    if network._mesh2d.mesh2d_face_z.size > 0:

    def clip_nodes(self, xnodes, ynodes, edge_nodes, polygon, keep_outside=False):
        """
        Method that returns a set of nodes that is clipped within a given polygon.

        Parameters
        ----------
        xnodes : 
            X-coordinates of mesh nodes
        ynodes : 
            Y-coordinates of mesh nodes
        edge_nodes : 
            node ids of edges. These are assumed in 1-based numbering
        polygon : Polygon, Multipolygon or list of

        """

        # Find nodes within one of the polygons
        logger.info('Selecting nodes within polygon.')
        index = np.zeros(len(xnodes), dtype=bool)
        for poly in geometry.as_polygon_list(polygon):
            index = index | geometry.points_in_polygon(np.c_[xnodes, ynodes], poly)
        if keep_outside:
            index = ~index

        # Filter edge nodes, keep the ones that are within a polygon with both points
        nodeids_in_polygon = np.where(index)[0] + 1
        edge_nodes = edge_nodes[np.isin(edge_nodes, nodeids_in_polygon).all(axis=1)]

        # Remove edges that intersect the clipping polygons. This is needed for when the clipping polygon is narrow,
        # So it crosses an edge without containing one of the edge nodes
        logger.info('Creating LineString for each edge.')
        edges = [LineString(line) for line in np.c_[xnodes, ynodes][edge_nodes - 1]]

        logger.info('Checking for intersections.')
        isect = np.zeros(len(edge_nodes), dtype=bool)
        for poly in geometry.as_polygon_list(polygon):
            prepped = prep(poly.exterior)
            isect = isect | np.array([prepped.intersects(e) for e in edges])
            for interior in poly.interiors:
                prepped = prep(interior)
                isect = isect | np.array([prepped.intersects(e) for e in edges])
        edge_nodes = edge_nodes[~isect]

        # Remove edges that have a 1 count, until none are left
        logger.info('Remove edges with only a single node within the clip area.')
        unique, count = np.unique(edge_nodes, return_counts=True)
        while any(count == 1):
            count1nodes = unique[count == 1]
            edge_nodes = edge_nodes[~np.isin(edge_nodes, count1nodes).any(axis=1)]
            unique, count = np.unique(edge_nodes, return_counts=True)

        # Select remaining nodes
        xnodes = xnodes[unique - 1]
        ynodes = ynodes[unique - 1]

        # Make mapping for new id's
        new_id_mapping = {old_id: new_id + 1 for new_id, old_id in enumerate(unique)}
        edge_nodes = np.reshape([new_id_mapping[old_id] for old_id in edge_nodes.ravel()], edge_nodes.shape)

        return xnodes, ynodes, edge_nodes

    def clip_mesh_by_polygon(self, polygon, outside=True):
        """
        Removes part of the existing mesh within the given polygon.

        Parameters
        ----------
        polygon : list, Polygon, MultiPolygon
            A polygon, multi-polygon or list of multiple polygons or multipolygons.
            The faces within the polygons are removed.

        self.meshgeom.set_values('nodex', xnodes)

        self.meshgeom.set_values('edge_nodes', np.ravel(edge_nodes).tolist())
        self.meshgeom.set_values('face_nodes', np.ravel(face_nodes).tolist())

    def clean_nodes(self, xnodes, ynodes, edge_nodes, face_nodes):
        """
        Method to clean the nodes. Edges that do not form a cell are deleted.
        Alternative method based on checking tuple pairs
        """

        # Select nodes the occur in any face
        node_selection = np.unique(face_nodes)
        node_selection = node_selection[node_selection!=-999]
        # If no nodes (because no faces), return None
        if not node_selection.any():
            return None

        # Remove all edges that are not in this selection
        edge_nodes = edge_nodes[np.isin(edge_nodes, node_selection).all(axis=1)]

        # Remove all segments that are not part of any face
        # Direction is irrelevant, sort to compare in ascending order
        # Convert to a contiguous array for faster comparison
        sorted_edges = np.ascontiguousarray(np.sort(edge_nodes, axis=1)).view(np.dtype((np.void, edge_nodes.dtype.itemsize * edge_nodes.shape[1])))
        idx = np.zeros(len(sorted_edges), dtype=bool)

        # Get combinations of face nodes that can form an edge
        combs = list(combinations(range(len(face_nodes[0])), 2))
        for comb in combs:
            # Get the first, second, third, etc. possible edge of every face
            arr = face_nodes[:, comb]
            # Remove nodata (so for example the non-existing fourth edge of a triangular face)
            arr = arr[(arr != -999).all(axis=1)]
            arr = np.sort(arr, axis=1)
            # Convert to a contiguous array for faster comparison
            sorted_face_edges = np.ascontiguousarray(arr).view(np.dtype((np.void, arr.dtype.itemsize * arr.shape[1])))
            # Check if face edges are in edges
            idx |= np.isin(sorted_edges, sorted_face_edges)[:, 0]

        # Select the nodes that are present in faces
        edge_nodes = edge_nodes[idx]

        xnodes = xnodes[node_selection - 1]
        ynodes = ynodes[node_selection - 1]

            # Make mapping for new id's
        new_id_mapping = {old_id: new_id + 1 for new_id, old_id in enumerate(node_selection)}    

        edge_nodes = np.reshape([-999 if old_id == -999 else new_id_mapping[old_id] for old_id in edge_nodes.ravel()], edge_nodes.shape)
        face_nodes = np.reshape([-999 if old_id == -999 else new_id_mapping[old_id] for old_id in face_nodes.ravel()], face_nodes.shape)

        return xnodes, ynodes, edge_nodes, face_nodes

    def altitude_from_raster(self, rasterpath, where='face', stat='mean', missing='default'):
        """
        Method to determine level of nodes

        This function works faster for large amounts of cells, since it does not
        draw polygons but checks for nearest neighbours (voronoi) based
        on interpolation.

        Note that the raster is not clipped. Any values outside the bounds are
        also taken into account.
        """

        # Select points on faces or nodes
        logger.info('Creating GeoDataFrame of cell faces.')
        xy = np.c_[self.meshgeom.get_values(f'{where}x'), self.meshgeom.get_values(f'{where}y')]
        cells = self.meshgeom.get_faces()
        facedata = gpd.GeoDataFrame(geometry=[Polygon(cell) for cell in cells])   

        if where == 'face':
            # Get raster statistics
            facedata.index = np.arange(len(xy), dtype=np.uint32) + 1
            # facedata.to_file('test_face.shp')
            facedata['crds'] = [cell for cell in cells]

            df = geometry.raster_stats_fine_cells(rasterpath, facedata, stats=[stat])
            # Get z values
            zvalues = df[stat].values

        elif where == 'node':

            logger.info('Generating voronoi polygons around cell centers for determining raster statistics.')

            # Creat voronoi polygon
            # Add border to limit polygons
            border = box(xy[:, 0].min(), xy[:, 1].min(), xy[:, 0].max(), xy[:, 1].max()).buffer(1000).exterior
            borderpts = [border.interpolate(dist).coords[0] for dist in np.linspace(0, border.length, max(20, border.length//100))]
            vor = Voronoi(points=xy.tolist()+borderpts)
            clippoly = facedata.union_all()
            # Get lines
            lines = []
            for poly in geometry.as_polygon_list(clippoly):
                lines.append(poly.exterior)
                lines.extend([line for line in poly.interiors])
            linesprep = prep(MultiLineString(lines))
            clipprep = prep(clippoly)

            # Collect polygons
            data = []
            for (pr, (i, pt)) in zip(vor.point_region, enumerate(xy)):
                region = vor.regions[pr]
                if pr == -1:
                    break
                while -1 in region:
                    region.remove(-1)
                if len(region) < 3:
                    continue
                crds = vor.vertices[region]
                if clipprep.intersects(Point(pt)):
                    poly = Polygon(crds)
                    if linesprep.intersects(poly):
                        poly = poly.intersection(clippoly)
                        if isinstance(poly, MultiPolygon):
                            poly = poly.buffer(0.001)
                        if isinstance(poly, MultiPolygon):
                            logger.warning('Got multipolygon when clipping voronoi polygon. Only adding coordinates for largest of the polygons.')
                            poly = poly[np.argmax([p.area for p in geometry.as_polygon_list(poly)])]
                        crds = np.vstack(poly.exterior.coords[:])
                    data.append({'geometry': poly, 'crds': crds})

            # Limit to model extend
            facedata = gpd.GeoDataFrame(data)
            facedata.index=np.arange(len(xy), dtype=np.uint32) + 1
            # facedata.drop('crds', axis=1).to_file('test_node.shp')
            # Get raster statistics
            df = geometry.raster_stats_fine_cells(rasterpath, facedata, stats=[stat])
            # Get z values
            zvalues = df[stat].values

        else:
            raise NotImplementedError()


        # If there are no NaN's, return the answer
        isnan = np.isnan(zvalues)
        if not isnan.any():
            # Set values to mesh geometry
            self.meshgeom.set_values(f'{where}z', zvalues)
            return None

        # If interpolation or missing, but all are NaN, raise error.
        elif isnan.all() and missing in ['nearest', 'interpolation']:
            raise ValueError('Only NaN values found, interpolation or nearest not possible.')

        # Fill missing values
        if missing.lower() == 'default':
            # With default value
            zvalues[isnan] = self.fill_value_z

        elif isinstance(missing, (float, int)):
            # With a given number
            zvalues[isnan] = missing

        elif missing.lower() == 'nearest':
            # By looking for the nearest value in the grid
            # Create a KDTree of the known points
            tree = KDTree(data=xy[~isnan])
            idx = tree.query(x=xy[isnan])[1]
            zvalues[isnan] = zvalues[~isnan][idx]

        elif missing.lower() == 'interpolation':
            # By interpolating
            isnan = np.isnan(zvalues)
            interp = LinearNDInterpolator(xy[~isnan], zvalues[~isnan], fill_value=self.fill_value_z)
            zvalues[isnan] = interp(xy[isnan])

        else:
            raise ValueError(f'{missing} not recognized. Choose \'default\', \'nearest\', \'interpolation\' or a number with which to fill the missing data.')

        # Set values to mesh geometry
        self.meshgeom.set_values(f'{where}z', zvalues)

    def generate_grid(self, x0, y0, dx, dy, ncols, nrows, clipgeo=None, rotation=0):
    """
    Generate rectangular grid based on the origin (x0, y0) the cell sizes (dx, dy),
    the number of columns and rows (ncols, nrows) and a rotation in degrees (default=0)
    A geometry (clipgeo) can be given to clip the grid.
    """

    # Generate x and y spacing
    x = np.linspace(x0, x0 + dx * (ncols), ncols+1)
    y = np.linspace(y0, y0 + dy * (nrows), nrows+1)

    # Get nodes
    xnodes, ynodes = np.meshgrid(x, y)

    # Rotate
    if rotation != 0:
        self.rotated = True
        xnodes, ynodes = geometry.rotate_coordinates((x0, y0), np.radians(rotation), xnodes, ynodes)

    # Get nodes as list
    nodes = {crd: i+1 for i, crd in enumerate(zip(xnodes.ravel(), ynodes.ravel()))}

    # Create segments
    segments = list(zip(
        zip(*np.c_[xnodes[:, :-1].ravel(), ynodes[:, :-1].ravel()].T),
        zip(*np.c_[xnodes[:, 1:].ravel(), ynodes[:, 1:].ravel()].T)
    ))
    segments += list(zip(
        zip(*np.c_[xnodes[1:, :].ravel(), ynodes[1:, :].ravel()].T),
        zip(*np.c_[xnodes[:-1, :].ravel(), ynodes[:-1, :].ravel()].T)
    ))

    # Get links
    edge_nodes = np.asarray([(nodes[s[0]], nodes[s[1]]) for s in segments])

    # Rvael the nodes
    xnodes = xnodes.ravel()
    ynodes = ynodes.ravel()

    # Add to mesh
    dimensions = meshgeomdim()
    dimensions.dim = 2
    geometries = meshgeom(dimensions)

    # Clip
    if clipgeo is not None:
        xnodes, ynodes, edge_nodes = self.clip_nodes(xnodes, ynodes, edge_nodes, clipgeo)

    # Update dimensions
    dimensions.numnode = len(xnodes)
    dimensions.numedge = len(edge_nodes)

    # Add nodes and links
    geometries.set_values('nodex', xnodes)
    geometries.set_values('nodey', ynodes)
    geometries.set_values('edge_nodes', np.ravel(edge_nodes).tolist())

    # Determine what the cells are
    self._find_cells(geometries, dll_path=self.dll_path)

    # Clip
    if clipgeo is not None:
        # Clean nodes, this function deletes nodes based on no longer existing cells
        face_nodes = geometries.get_values('face_nodes', as_array=True)
        cleaned = self.clean_nodes(xnodes, ynodes, edge_nodes, face_nodes)
        # If cleaning leaves no whole cells, return None
        if cleaned is None:
            return None
        # Else, get the arguments from the returned tuple
        else:
            xnodes, ynodes, edge_nodes, face_nodes = cleaned

    # Update dimensions
    dimensions.numnode = len(xnodes)
    dimensions.numedge = len(edge_nodes)
    dimensions.maxnumfacenodes = 4

    # Add nodes and links
    geometries.set_values('nodex', xnodes)
    geometries.set_values('nodey', ynodes)
    geometries.set_values('edge_nodes', np.ravel(edge_nodes).tolist())
    geometries.set_values('face_nodes', np.ravel(face_nodes).tolist())

    # Add to mesh
    self.meshgeom.add_from_other(geometries)

    class GeometryList(GeometryListMK):
    """meshkernel GeometryList that converts from and to shapely geometries.

    The coordinates of all parts are stored in one array. Parts of a multi-geometry
    are separated by ``geometry_separator`` (-999), the holes of a polygon by
    ``inner_outer_separator`` (-998).
    """

    @classmethod
    def from_geometry(cls, geometry: BaseGeometry) -> Self:
        """Create a GeometryList from a shapely geometry.

        Parameters
        ----------
        geometry : BaseGeometry
            A Point, LineString or Polygon, or a multi-geometry of those.

        Returns
        -------
        GeometryList
            The coordinates of the geometry.

        Raises
        ------
        TypeError
            If the geometry is of another type, e.g. a GeometryCollection.
        """
        if isinstance(geometry, Polygon):
            return cls.from_polygon(geometry)
        elif isinstance(geometry, MultiPolygon):
            return cls.from_multipolygon(geometry)
        elif isinstance(geometry, LineString):
            return cls.from_linestring(geometry)
        elif isinstance(geometry, MultiLineString):
            return cls.from_multilinestring(geometry)
        elif isinstance(geometry, Point):
            return cls.from_point(geometry)
        elif isinstance(geometry, MultiPoint):
            return cls.from_multipoint(geometry)
        else:
            raise TypeError(f"Geometry type {type(geometry)} not understood.")

    @classmethod
    def _from_simple(cls, geometry: LineString | Point) -> Self:
        # Extract coordinates from geometry
        x_crds, y_crds = np.array(geometry.coords[:]).T
        gl = cls(x_coordinates=x_crds, y_coordinates=y_crds)
        return gl

    @classmethod
    def from_linestring(cls, linestring: LineString) -> Self:
        """Create a GeometryList from a LineString."""
        return cls._from_simple(linestring)

    @classmethod
    def from_point(cls, point: Point) -> Self:
        """Create a GeometryList from a Point."""
        return cls._from_simple(point)

    @classmethod
    def from_polygon(cls, polygon: Polygon) -> Self:
        """Create a GeometryList from a Polygon: the exterior, then each hole after a separator."""
        # Create a list of coordinate lists
        # Add exterior
        x_ext, y_ext = np.array(polygon.exterior.coords[:]).T
        x_crds = [x_ext]
        y_crds = [y_ext]

        if not hasattr(cls, "inner_outer_separator"):
            cls.inner_outer_separator = -998

        # Add interiors, seperated by inner_outer_separator
        for interior in polygon.interiors:
            x_int, y_int = np.array(interior.coords[:]).T
            x_crds.append([cls.inner_outer_separator])
            x_crds.append(x_int)
            y_crds.append([cls.inner_outer_separator])
            y_crds.append(y_int)
        gl = cls(
            x_coordinates=np.concatenate(x_crds), y_coordinates=np.concatenate(y_crds)
        )
        return gl

    @classmethod
    def _from_multigeometry(
        cls, multigeometry: MultiLineString | MultiPolygon | MultiPoint
    ) -> Self:
        x_crds = []
        y_crds = []
        cls.inner_outer_separator = -998
        cls.geometry_separator = -999
        # Create a GeometryList for every polygon in the multipolygon
        for i, geometry in enumerate(multigeometry.geoms):
            gl = cls.from_geometry(geometry)
            # Add geometry seperator for every geometrylist, except the first one
            if i != 0:
                x_crds.append([cls.geometry_separator])
                y_crds.append([cls.geometry_separator])
            # Add coordinates of polygon GeometryList
            x_crds.append(gl.x_coordinates)
            y_crds.append(gl.y_coordinates)

        gl = cls(
            x_coordinates=np.concatenate(x_crds), y_coordinates=np.concatenate(y_crds)
        )
        return gl

    @classmethod
    def from_multipolygon(cls, multipolygon: MultiPolygon) -> Self:
        """Create a GeometryList from a MultiPolygon, with a separator between the polygons."""
        return cls._from_multigeometry(multipolygon)

    @classmethod
    def from_multilinestring(cls, multilinestring: MultiLineString) -> Self:
        """Create a GeometryList from a MultiLineString, with a separator between the lines."""
        return cls._from_multigeometry(multilinestring)

    @classmethod
    def from_multipoint(cls, multipoint: MultiPoint) -> Self:
        """Create a GeometryList from a MultiPoint, with a separator between the points."""
        return cls._from_multigeometry(multipoint)

    def _to_polygon(
        self, geometries: list[GeometryListMK], is_multi: bool
    ) -> Polygon | MultiPolygon:
        polygons = []
        for geometry in geometries:
            parts = [
                np.stack([p.x_coordinates, p.y_coordinates], axis=1)
                for p in split_by(geometry, self.inner_outer_separator)
            ]
            polygons.append(Polygon(shell=parts[0], holes=parts[1:]))
        if is_multi:
            return MultiPolygon(polygons)
        else:
            return polygons[0]

    def _to_linestring(
        self, geometries: list[GeometryListMK], is_multi: bool
    ) -> LineString | MultiLineString:
        linestrings = [
            LineString(np.stack([p.x_coordinates, p.y_coordinates], axis=1))
            for p in geometries
        ]
        if is_multi:
            return MultiLineString(linestrings)
        else:
            return linestrings[0]

    def _to_points(
        self, geometries: list[GeometryListMK], is_multi: bool
    ) -> Point | MultiPoint:
        points = [
            Point(np.stack([p.x_coordinates, p.y_coordinates], axis=1))
            for p in geometries
        ]
        if is_multi:
            return MultiPoint(points)
        else:
            return points[0]

    def to_geometry(
        self,
    ) -> Polygon | MultiPolygon | LineString | MultiLineString | Point | MultiPoint:
        """Convert the GeometryList to a shapely geometry.

        The first part determines the type: a closed ring of at least 4 coordinates
        is a polygon, more than one coordinate a line, otherwise a point. With more
        than one part, the result is a multi-geometry.

        Returns
        -------
        Polygon, MultiPolygon, LineString, MultiLineString, Point or MultiPoint
            The geometry.
        """
        geometries = [
            geo
            for geo in split_by(self, self.geometry_separator)
            if geo.x_coordinates.size > 0
        ]
        is_multi = len(geometries) > 1

        # The first geometry determines the type
        exterior = split_by(geometries[0], self.inner_outer_separator)[0]
        x, y = exterior.x_coordinates, exterior.y_coordinates
        # A ring has at least 4 coordinates, and its first and last coordinate are equal
        if len(x) >= 4 and x[0] == x[-1] and y[0] == y[-1]:
            return self._to_polygon(geometries, is_multi)
        if len(x) > 1:
            return self._to_linestring(geometries, is_multi)
        return self._to_points(geometries, is_multi)

    @property
    def geoms(
        self,
    ) -> Iterator[
        Polygon | MultiPolygon | LineString | MultiLineString | Point | MultiPoint
    ]:
        """Yield each part, split at the geometry separator, as a shapely geometry."""
        for geometrylist in split_by(self, self.geometry_separator):
            # yield as shapely geometry
            yield GeometryList(**geometrylist.__dict__).to_geometry()

    The columns ``branch_id`` and ``branch_offset`` of ``geometries`` are filled in, and

    least 0.1 m (or half the branch length) from both branch ends. Geometries without a

        - ``"overal"``: the nearest branch to the geometry as a whole;
        - ``"centroid"``: the nearest branch to the centroid;
        - ``"ends"``: the smallest mean distance to both ends of a line;
        - ``"intersecting"``: each branch that intersects the geometry; the offset is
          at the centroid of the intersection, and a geometry intersecting more
          branches gets the last one.
    maxdist : float, optional
        Maximum distance to a branch. Not used for ``"intersecting"``. Defaults to 5.

    Raises
    ------
    NotImplementedError
        If the method is not one of the above.
    """
    # Check if method is in allowed methods
    allowed_methods = ["intersecting", "overal", "centroid", "ends"]
    if method not in allowed_methods:
        raise NotImplementedError(f'Method "{method}" not implemented.')

    # Add columns if not present
    if "branch_id" not in geometries.columns:
        geometries["branch_id"] = ""
    if "branch_offset" not in geometries.columns:
        geometries["branch_offset"] = np.nan

    if method == "intersecting":
        # Determine intersection geometries per branch
        sindex = geometries.sindex
        for branch_id, branchgeo in zip(branches.index, branches.geometry):
            intersecting = geometries.iloc[
                sindex.query(branchgeo, predicate="intersects")
            ]

            # For each geometrie, determine offset along branch
            for geometry_id, geometry in zip(intersecting.index, intersecting.geometry):
                # Determine distance of profile line along branch
                geometries.at[geometry_id, "branch_id"] = branch_id

                # Calculate offset
                mindist = min(0.1, branchgeo.length / 2.0)
                offset = round(
                    branchgeo.project(branchgeo.intersection(geometry).centroid), 3
                )
                offset = max(mindist, min(branchgeo.length - mindist, offset))
                geometries.at[geometry_id, "branch_offset"] = offset

    else:
        sindex = branches.sindex
        # the code of each geometry, for the log messages
        has_code = "code" in geometries.columns
        codes = [None] * len(geometries)
        if has_code:
            codes = geometries["code"].tolist()
        # In case of looking for the nearest, it is easier to iteratie over the geometries instead of the branches
        for geometry_id, geometry, code in zip(
            geometries.index, geometries.geometry, codes

            nearidx = sindex.query(geometry, predicate="dwithin", distance=maxdist)
            selectie = branches.iloc[nearidx]

            if method == "overal":
                # Determine distances to branches
                dist = selectie.distance(geometry)
            elif method == "centroid":
                # Determine distances to branches
                dist = selectie.distance(geometry.centroid)
            elif method == "ends":
                # Since a culvert can cross a channel, it is
                crds = geometry.coords[:]
                dist = (
                    selectie.distance(Point(*crds[0]))
                    + selectie.distance(Point(*crds[-1]))
                ) * 0.5

            # Enforce a point
            geo = geometry
            if not isinstance(geo, Point):
                geo = geo.centroid

            # Log-friendly name with coordinates
            geom_name = f"at x={geo.x:.4f} and y={geo.y:.4f}"
            if has_code:
                geom_name = f"for {code} {geom_name}"

            # Determine nearest
            if dist.min() < maxdist:
                branchidxmin = dist.idxmin()
                geometries.at[geometry_id, "branch_id"] = branchidxmin

                nbranches = (dist < maxdist).sum()
                if nbranches > 1:

                        "Centroid %s has %d branches with maxdist=%s and method=%s, chosing closest branch %s",
                        geom_name,
                        nbranches,
                        maxdist,

                branchgeo = branches.geometry[branchidxmin]
                mindist = min(0.1, branchgeo.length / 2.0)

                    mindist,
                    min(branchgeo.length - mindist, round(branchgeo.project(geo), 3)),
                )
                geometries.at[geometry_id, "branch_offset"] = offset
            else:
                logger.info(
                    "Centroid %s has 0 branches with maxdist=%s and method=%s",
                    geom_name,
                    maxdist,
                    method,
                )

    unique_nodes, inverse = np.unique(nodes, axis=0, return_inverse=True)

    return gpd.GeoDataFrame(

    network: Network,
    crs: int = 28992,
    mesh1d_kwargs: dict | None = None,
    mesh2d_kwargs: dict | None = None,
    links1d2d_kwargs: dict | None = None,

    """Show the 1D mesh, the 2D mesh, the 1D2D links and the 2D face levels on a folium map.

    Each part is a separate layer; empty parts are left out. The map zooms to the
    extent of all layers.

    Parameters

    network : Network
        Network to show.
    crs : int, optional
        EPSG code of the network coordinates. Defaults to 28992 (RD New).
    mesh1d_kwargs, mesh2d_kwargs, links1d2d_kwargs : dict, optional
        Line style of each layer, with the keys ``color`` and ``weight`` (``lw`` is
        accepted for ``weight``). Defaults to red, blue and black lines.

        ``vmax``, ``cmap``, ``label`` (legend caption) and ``radius`` (in points).
        Missing keys get defaults; ``vmin`` and ``vmax`` default to the range of the
        levels. Defaults to None: no face levels.

    Returns

    folium.Map

        mesh1d_kwargs = {"color": "#d62728", "weight": 1}
    if mesh2d_kwargs is None:
        mesh2d_kwargs = {"color": "#1f77b4", "weight": 0.5}
    if links1d2d_kwargs is None:
        links1d2d_kwargs = {"color": "black", "weight": 1}

    for d in [mesh1d_kwargs, mesh2d_kwargs, links1d2d_kwargs]:
        if "lw" in d and "weight" not in d:
            d["weight"] = d.pop("lw")

    if background:

    else:
        m = folium.Map(location=[0, 0], zoom_start=5, tiles=None, prefer_canvas=True)
    all_bounds = []
    nodes1d = faces2d = None

    def edges_to_lines(
        nodes: np.ndarray, edge_nodes: np.ndarray
    ) -> LineString | np.ndarray:
        coords = nodes[edge_nodes]
        return shapely.linestrings(
            coords.reshape(-1, 2),
            indices=np.repeat(np.arange(len(edge_nodes)), 2),
        )

        geoms: LineString | np.ndarray, name: str, color: str, weight: float
    ) -> None:

            if face_z_kwargs["vmin"] is not None

            name="Face level",
            opacity=0.8,
        ).add_to(m)

            colors=[
                mcolors.to_hex(c)

    zonal ¶

    Zonal statistics computed with exactextract.

    exactextract computes the exact fraction of every pixel covered by a zone. Two of its arguments, applied to every statistic, use that fraction:

    • min_coverage_frac selects the pixels: those covered less are ignored.
    • coverage_weight sets how they count: "none" counts every selected pixel fully, "fraction" by the fraction of it the zone covers.

    The defaults, min_coverage_frac=0 and coverage_weight="fraction", give exact coverage weighting: every touched pixel counts by the fraction of it the zone covers. coverage_weight="none" counts every touched pixel fully, like all_touched=True in GDAL and rasterstats; adding min_coverage_frac=0.5 counts only pixels that are at least half covered, close to their default pixel-centre rule.

    Raster files are read through rasterio. exactextract reads windows of at most max_cells_in_memory cells, so rasters may be larger than memory.

    class ZonalCache:
    """Reuse the results of identical :func:`zonal_stats` and :func:`zonal_category_counts` calls.

    Only calls on raster files are cached. A result is reused only when the zones (index, CRS and geometries), the
    raster file (path, size and modification time), and every argument that
    affects the result are the same. Arrays and open datasets are never cached.
    Results are copied in and out, so changing a returned result does not
    change the cache. Entries stay until :meth:`clear`.
    """

    # Arguments that do not affect the result.
    _NOT_IN_KEY = frozenset(
        {"zones", "raster", "strategy", "max_cells_in_memory", "progress"}
    )

    def __init__(self) -> None:
        self._results: dict[tuple[object, ...], pd.DataFrame] = {}

    def __len__(self) -> int:
        """Return the number of cached results."""
        return len(self._results)

    def clear(self) -> None:
        """Remove all cached results."""
        self._results.clear()

    def zonal_stats(
        self,
        zones: Zones,
        raster: Raster,
        statistics: str | Sequence[str],
        **kwargs: Unpack[ZonalOptions],
    ) -> pd.DataFrame:
        """:func:`zonal_stats`, reusing an identical earlier result."""
        if isinstance(statistics, str):
            statistics = [statistics]
        return self._get(
            zonal_stats, zones, raster, statistics=tuple(statistics), **kwargs
        )

    def zonal_category_counts(
        self, zones: Zones, raster: Raster, **kwargs: Unpack[ZonalOptions]
    ) -> pd.DataFrame:
        """:func:`zonal_category_counts`, reusing an identical earlier result."""
        return self._get(zonal_category_counts, zones, raster, **kwargs)

    def _get(
        self,
        function: Callable[..., pd.DataFrame],
        zones: Zones,
        raster: Raster,
        **kwargs: object,
    ) -> pd.DataFrame:
        if not isinstance(raster, (str, os.PathLike)):
            return function(zones, raster, **kwargs)
        try:
            path = Path(raster).resolve()
            status = path.stat()
        except OSError:  # not a local file, such as a /vsicurl/ URL
            return function(zones, raster, **kwargs)

        arguments = inspect.signature(function).bind(zones, raster, **kwargs)
        arguments.apply_defaults()
        key = (
            function.__name__,
            str(path),
            status.st_size,
            status.st_mtime_ns,
            _zones_key(zones),
            *(
                (name, _crs_key(value) if name == "raster_crs" else value)
                for name, value in arguments.arguments.items()
                if name not in self._NOT_IN_KEY
            ),
        )
        if key in self._results:
            logger.debug("Reusing %s of %s.", function.__name__, path.name)
        else:
            self._results[key] = function(zones, raster, **kwargs)

    zones: Zones,
    raster: Raster,
    *,
    strategy: Strategy,
    min_coverage_frac: float = 0.0,
    coverage_weight: CoverageWeight = "fraction",
    affine: Affine | None = None,
    raster_crs: CRS | str | None = None,
    nodata: float | None = None,
    max_cells_in_memory: int = 30_000_000,
    progress: Progress = False,
) -> pd.DataFrame:
    """Count the cells of each raster class within every polygonal zone.

    Parameters are those of :func:`zonal_stats`. With
    ``coverage_weight="fraction"``, partly covered cells count for their covered
    fraction, so counts are not whole numbers.

    Returns
    -------
    pandas.DataFrame
        One float column per class code, aligned to ``zones.index``. Integer
        class codes are ints. Classes absent from a zone count zero.
    """
    result = _extract(
        zones,
        [raster],
        ["unique", "frac", "count"],
        strategy=strategy,
        min_coverage_frac=min_coverage_frac,
        coverage_weight=coverage_weight,
        affine=affine,
        raster_crs=raster_crs,
        nodata=nodata,
        max_cells_in_memory=max_cells_in_memory,
        progress=progress,
    )
    rows = [
        {
            int(code) if float(code).is_integer() else code: share * count
            for code, share in zip(classes.tolist(), np.asarray(frac).tolist())
        }
        if isinstance(classes, np.ndarray)  # None for zones outside the raster
        else {}
        for classes, frac, count in zip(
            result["unique"], result["frac"], result["count"]
        )
    ]
    counts = pd.DataFrame(rows, index=zones.index, dtype=np.float64)
    counts = counts.sort_index(axis=1).fillna(0.0)
    # Unweighted counts are whole cells; drop floating-point noise.
    return counts.round() if coverage_weight == "none" else counts

    def zonal_stats_per_raster(
    zones: Zones,
    rasters: Sequence[Raster],
    statistic: str,
    *,
    strategy: Strategy,
    min_coverage_frac: float = 0.0,
    coverage_weight: CoverageWeight = "fraction",
    affine: Affine | None = None,
    raster_crs: CRS | str | None = None,
    nodata: float | None = None,
    max_cells_in_memory: int = 30_000_000,
    progress: Progress = False,
) -> pd.DataFrame:
    """Calculate one scalar raster statistic for every zone, in each raster.

    Faster than calling :func:`zonal_stats` per raster, as for a time series of
    rasters: exactextract handles up to ``_RASTERS_PER_CALL`` rasters at once.

    Parameters
    ----------
    rasters : list of pathlib.Path, str, rasterio dataset, or array
        Raster files, open rasterio datasets, or two-dimensional arrays that
        share ``affine`` and ``nodata``. Lazy arrays, such as xarray's, are
        read only when their turn comes, so the list may hold more than fits in
        memory. Only band 1 of a file is used.
    statistic : str
        One scalar exactextract statistic, such as ``"mean"``.

    Other parameters are those of :func:`zonal_stats`.

    Returns
    -------
    pandas.DataFrame
        One float column per raster, numbered by position in ``rasters``, and
        aligned to ``zones.index``.
    """
    rasters = list(rasters)
    results = []
    with ExitStack() as stack:
        if progress is True:
            progress = _progress_bar(stack)
        for start in range(0, len(rasters), _RASTERS_PER_CALL):
            chunk = rasters[start : start + _RASTERS_PER_CALL]
            chunk_progress: Progress = False
            if progress:
                chunk_progress = _rescale_progress(
                    progress, start, len(chunk), len(rasters)
                )
            result = _extract(
                zones,
                chunk,
                [statistic],
                strategy=strategy,
                min_coverage_frac=min_coverage_frac,
                coverage_weight=coverage_weight,
                affine=affine,
                raster_crs=raster_crs,
                nodata=nodata,
                max_cells_in_memory=max_cells_in_memory,
                progress=chunk_progress,
            )
            results.append(result)
    result = pd.concat(results, axis=1).set_axis(range(len(rasters)), axis=1)
    return result.apply(pd.to_numeric, errors="coerce").astype(np.float64)
